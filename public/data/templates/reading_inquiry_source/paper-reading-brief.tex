\PageTitle{ESSAY EXAMPLE / FIRST READ}{建立初步判断}
\vspace{0.08em}
\SourceLine

\RuleHead{QUESTION}{Initial question}

{\bfseries\fontsize{17pt}{22pt}\selectfont
  何种方法能够将新增计算转化为持续进步？\par}

\vspace{0.16em}
{\small\color{muted}
  先处理原文，区分主张、证据与个人解释；暂不引入 AI 生成内容。}

\vspace{0.40em}
\begin{minipage}[t]{0.69\linewidth}
  \vspace{0pt}
  \RuleHead{TRACE}{Source notes}

  \SourceNote{开篇}{claim}{
    长期占优的，往往是能够持续利用新增计算的一般方法。}
  \vspace{0.55em}{\color{rule}\hrule height 0.4pt}\vspace{0.52em}
  \SourceNote{中段}{evidence}{
    棋类、语音与视觉呈现相似模式：手工知识改善早期性能，搜索与学习更能受益于规模扩张。}
  \vspace{0.55em}{\color{rule}\hrule height 0.4pt}\vspace{0.52em}
  \SourceNote{结尾}{direction}{
    重点不在手工编码复杂性，而在构造能够发现并吸收复杂性的过程。}

  \RuleHead{MY READ}{A tension worth keeping}
  \begin{tcolorbox}[openbox]
    {\bfseries 领域知识改善初始性能，也可能成为可扩展性的约束。}\par
    \vspace{0.35em}
    {\small\color{muted}
      个人解释；原文未提供因果识别。}
  \end{tcolorbox}
\end{minipage}\hfill
\begin{minipage}[t]{0.26\linewidth}
  \vspace{0pt}
  \RuleHead{AI OFF}{First pass}
  \begin{tcolorbox}[marginbox]
    {\footnotesize\color{muted}
      暂不生成结论，仅保留初读中的张力与问题。}

    \vspace{0.72em}
    \MarginQuestion{concept}{“一般方法”的操作性含义是什么？}
    \vspace{0.72em}
    \MarginQuestion{evidence}{例证识别了规律，还是支持因果解释？}
    \vspace{0.72em}
    \MarginQuestion{boundary}{领域先验应位于何处？}
  \end{tcolorbox}
\end{minipage}

\vfill
\RuleHead{STOP}{First-pass record}
{\small
  三处来源、一条解释、三个问题。初读到此为止。}

\PageTitle{ESSAY EXAMPLE / EXPAND}{展开候选解释}
\vspace{0.08em}
\SourceLine

\begin{minipage}[t]{0.69\linewidth}
  \vspace{0pt}
  \RuleHead{HANDOFF}{Inputs and constraints}
  \begin{tcolorbox}[openbox]
    {\small
      输入：三处来源、初步判断与未决问题。\par
      \color{muted}约束：仅生成候选，并标记原文依据、推断与待核验处。}
  \end{tcolorbox}

  \RuleHead{BRANCH 01}{Clarify the distinction}
  {\bfseries 候选解释：}\;
  核心区分或许不在“知识或计算”，而在知识被固化为答案，
  还是用于构造可持续搜索与学习的过程。\par
  \vspace{0.25em}
  {\footnotesize\color{muted}
    核验：原文支持的重述，还是新增分析框架？}

  \RuleHead{BRANCH 02}{Oppose the claim}
  {\bfseries 候选反例：}\;
  安全约束、物理结构与昂贵反馈可能长期依赖领域知识。\par
  \vspace{0.25em}
  {\footnotesize\color{muted}
    追问：这些知识提供答案，还是定义边界与反馈？}

  \RuleHead{BRANCH 03}{Transfer carefully}
  {\bfseries 候选迁移：}\;
  在模型、工具与 agent 工作流中，哪些结构属于临时脚手架，
  哪些构成长期接口？\par
  \vspace{0.25em}
  {\footnotesize\color{muted}
    标记：后续迁移；不属于 2019 年原文结论。}
\end{minipage}\hfill
\begin{minipage}[t]{0.26\linewidth}
  \vspace{0pt}
  \RuleHead{OPTIONAL}{AI margin}
  \begin{tcolorbox}[marginbox]
    {\footnotesize\color{muted}
      AI 仅扩展假设空间，不修改主笔记。}

    \vspace{0.70em}
    \MarginQuestion{clarify}{给出最小必要的概念区分。}
    \vspace{0.70em}
    \MarginQuestion{alternate}{提出可竞争的替代机制。}
    \vspace{0.70em}
    \MarginQuestion{falsify}{何种证据足以改变当前判断？}
    \vspace{0.70em}
    \MarginQuestion{provenance}{区分原文结论与后续迁移。}
  \end{tcolorbox}
\end{minipage}

\vfill
\RuleHead{SELECT}{Selected branch}
\begin{tcolorbox}[openbox]
  仅推进分支 01：核查“人类知识 / 可扩展方法”是否构成真实二分。
  其余分支暂不写回。
\end{tcolorbox}

\PageTitle{ESSAY EXAMPLE / WRITE BACK}{核验后的修订}
\vspace{0.30em}
\SourceLine

\RuleHead{KEEP}{Source-backed}
\begin{tcolorbox}[openbox,fontupper=\small]
  原文持续对比两条路线：直接编码领域知识；
  构造能够随计算扩展的搜索与学习方法。
\end{tcolorbox}

\RuleHead{REVISE}{My interpretation}
\begin{tcolorbox}[openbox,fontupper=\small]
  初读将其概括为“知识与计算的冲突”，区分过粗。
  更准确的问题是：知识被固化为答案，还是用于设计可持续学习、搜索与吸收反馈的过程？
\end{tcolorbox}

\RuleHead{LEAVE OPEN}{A present-day transfer}
\begin{tcolorbox}[openbox,fontupper=\small]
  对于当代模型与 agent 工作流，哪些结构属于临时脚手架，
  哪些属于长期接口、安全边界与反馈机制？这是原文未回答的延伸问题。
\end{tcolorbox}

\RuleHead{MODEL}{A relation worth redrawing}
\begin{tcolorbox}[marginbox]
  \centering
  {\small\sffamily
    domain knowledge
    \quad$\longrightarrow$\quad
    fixed answer \; / \; scalable interface
    \quad$\longleftarrow$\quad
    growing computation}
\end{tcolorbox}

\RuleHead{NEXT}{One question to carry}

{\bfseries\large
  在当前 AI 工作流中，哪些结构应保持稳定，哪些应随模型能力更新？\par}
